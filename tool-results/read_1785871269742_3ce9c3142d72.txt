   223→  name: "",
   224→  category: "Agriculture",
   225→  price: "",

   233→function ProductDialog({
   234→  open,
   235→  onOpenChange,
   236→  initial,
   237→  onSaved,
   238→}: {
   239→  open: boolean;
   240→  onOpenChange: (o: boolean) => void;
   241→  initial?: ProductFormState | null;
   242→  onSaved: () => void;
   243→}) {
   244→  const [form, setForm] = useState<ProductFormState>(initial ?? EMPTY_FORM);
   245→  const [saving, setSaving] = useState(false);
   246→
   247→  useEffect(() => {
   248→    setForm(initial ?? EMPTY_FORM);
   249→  }, [initial, open]);
   250→
   251→  const set = <K extends keyof ProductFormState>(k: K, v: ProductFormState[K]) =>
   252→    setForm((f) => ({ ...f, [k]: v }));
   253→
   254→  const submit = async (e: React.FormEvent) => {
   255→    e.preventDefault();
   256→    if (saving) return;
   257→    setSaving(true);
   258→    try {
   259→      const benefits = form.benefitsText
   260→        .split("\n")
   261→        .map((s) => s.trim())
   262→        .filter(Boolean);
   263→      const payload: ProductPayload = {
   264→        name: form.name.trim(),
   265→        category: form.category.trim() || "Agriculture",
   266→        price: Number(form.price) || 0,
   267→        dailyIncome: Number(form.dailyIncome) || 0,
   268→        description: form.description,
   269→        image: form.image || "/wheat.jpg",
   270→        benefits,
   271→        status: form.status,
   272→      };
   273→      if (form.id) {
   274→        await productsApi.update({ id: form.id, ...payload });
   275→        toast.success("Product updated");
   276→      } else {
   277→        await productsApi.create(payload);
   278→        toast.success("Product created");
   279→      }
   280→      onSaved();
   281→      onOpenChange(false);
   282→    } catch (err) {
   283→      toast.error(err instanceof Error ? err.message : "Failed to save product");
   284→    } finally {
   285→      setSaving(false);
   286→    }
   287→  };
   288→
   289→  return (
   290→    <Dialog open={open} onOpenChange={onOpenChange}>
   291→      <DialogContent className="max-h-[90vh] overflow-y-auto scroll-gold sm:max-w-lg">
   292→        <DialogHeader>
   293→          <DialogTitle className="flex items-center gap-2">
   294→            <Package className="size-5 text-gold-deep" />
   295→            {form.id ? "Edit Product" : "Add Product"}
   296→          </DialogTitle>
   297→          <DialogDescription>
   298→            Fill in the package details. Benefits are one per line.
   299→          </DialogDescription>
   300→        </DialogHeader>
   301→        <form onSubmit={submit} className="space-y-3">
   302→          <div className="space-y-1.5">
   303→            <Label htmlFor="p-name">Name</Label>
   304→            <Input
   305→              id="p-name"
   306→              value={form.name}
   307→              onChange={(e) => set("name", e.target.value)}
   308→              required
   309→            />
   310→          </div>
   311→          <div className="grid grid-cols-2 gap-3">
   312→            <div className="space-y-1.5">
   313→              <Label htmlFor="p-cat">Category</Label>
   314→              <Input
   315→                id="p-cat"
   316→                value={form.category}
   317→                onChange={(e) => set("category", e.target.value)}
   318→              />
   319→            </div>
   320→            <div className="space-y-1.5">
   321→              <Label htmlFor="p-status">Status</Label>
   322→              <Select
   323→                value={form.status}
   324→                onValueChange={(v) =>
   325→                  set("status", v as "AVAILABLE" | "UNAVAILABLE")
   326→                }
   327→              >
   328→                <SelectTrigger id="p-status" className="w-full">
   329→                  <SelectValue />
   330→                </SelectTrigger>
   331→                <SelectContent>
   332→                  <SelectItem value="AVAILABLE">Available</SelectItem>
   333→                  <SelectItem value="UNAVAILABLE">Unavailable</SelectItem>
   334→                </SelectContent>
   335→              </Select>
   336→            </div>
   337→          </div>
   338→          <div className="grid grid-cols-2 gap-3">
   339→            <div className="space-y-1.5">
   340→              <Label htmlFor="p-price">Price (ETB)</Label>
   341→              <Input
   342→                id="p-price"
   343→                type="number"
   344→                min={0}
   345→                value={form.price}
   346→                onChange={(e) => set("price", e.target.value)}
   347→                required
   348→              />
   349→            </div>
   350→            <div className="space-y-1.5">
   351→              <Label htmlFor="p-daily">Daily income (ETB)</Label>
   352→              <Input
   353→                id="p-daily"
   354→                type="number"
   355→                min={0}
   356→                value={form.dailyIncome}
   357→                onChange={(e) => set("dailyIncome", e.target.value)}
   358→                required
   359→              />
   360→            </div>
   361→          </div>
   362→          <div className="space-y-1.5">
   363→            <Label htmlFor="p-desc">Description</Label>
   364→            <Textarea
   365→              id="p-desc"
   366→              rows={3}
   367→              value={form.description}
   368→              onChange={(e) => set("description", e.target.value)}
   369→            />
   370→          </div>
   371→          <div className="space-y-1.5">
   372→            <Label htmlFor="p-ben">Benefits (one per line)</Label>
   373→            <Textarea
   374→              id="p-ben"
   375→              rows={4}
   376→              value={form.benefitsText}
   377→              onChange={(e) => set("benefitsText", e.target.value)}

   379→            />

   381→          <div className="space-y-1.5">
   382→            <Label>Image</Label>
   383→            <ImageUpload
   384→              value={form.image}

   386→            />
   387→            <div className="flex items-center gap-2 text-xs text-muted-foreground">
   388→              <ImageIcon className="size-3.5" />
   389→              Upload a new image or keep the default wheat image.
   390→            </div>
   391→          </div>
   392→          <DialogFooter>
   393→            <Button
   394→              type="button"
   395→              variant="outline"
   396→              onClick={() => onOpenChange(false)}
   397→            >
   398→              Cancel
   399→            </Button>
   400→            <Button
   401→              type="submit"
   402→              disabled={saving}
   403→              className="bg-gold-gradient text-primary-foreground shadow hover:opacity-90 rounded-full"
   404→            >
   405→              {saving ? <Loader2 className="size-4 animate-spin" /> : null}
   406→              {form.id ? "Save Changes" : "Create Product"}
   407→            </Button>
   408→          </DialogFooter>
   409→        </form>
   410→      </DialogContent>
   411→    </Dialog>
   412→  );

   415→function ProductsTab() {
   416→  const [products, setProducts] = useState<ProductPublic[]>([]);
   417→  const [loading, setLoading] = useState(true);
   418→  const [dialogOpen, setDialogOpen] = useState(false);
   419→  const [editing, setEditing] = useState<ProductFormState | null>(null);
   420→  const [deletingId, setDeletingId] = useState<string | null>(null);
   421→
   422→  const load = async () => {
   423→    setLoading(true);
   424→    try {
   425→      const res = await productsApi.list(true);
   426→      setProducts(res.products);
   427→    } catch (err) {
   428→      toast.error(err instanceof Error ? err.message : "Failed to load products");
   429→    } finally {
   430→      setLoading(false);
   431→    }
   432→  };
   433→
   434→  useEffect(() => {
   435→    void load();
   436→  }, []);
   437→
   438→  const openAdd = () => {
   439→    setEditing(null);
   440→    setDialogOpen(true);
   441→  };
   442→  const openEdit = (p: ProductPublic) => {
   443→    setEditing({
   444→      id: p.id,
   445→      name: p.name,
   446→      category: p.category,
   447→      price: String(p.price),
   448→      dailyIncome: String(p.dailyIncome),
   449→      description: p.description,
   450→      image: p.image,
   451→      benefitsText: p.benefits.join("\n"),
   452→      status: p.status,
   453→    });
   454→    setDialogOpen(true);
   455→  };
   456→
   457→  const remove = async (id: string) => {
   458→    setDeletingId(id);
   459→    try {
   460→      await productsApi.remove(id);
   461→      toast.success("Product deleted");
   462→      await load();
   463→    } catch (err) {
   464→      toast.error(err instanceof Error ? err.message : "Failed to delete");
   465→    } finally {
   466→      setDeletingId(null);
   467→    }
   468→  };
   469→
   470→  return (
   471→    <div className="space-y-4">
   472→      <div className="flex items-center justify-between">
   473→        <h3 className="font-bold text-foreground">All Products</h3>
   474→        <Button
   475→          onClick={openAdd}
   476→          className="bg-gold-gradient text-primary-foreground shadow hover:opacity-90 rounded-full"
   477→        >
   478→          <Plus className="size-4" /> Add Product
   479→        </Button>
   480→      </div>
   481→      <Card className="p-0">
   482→        {loading ? (
   483→          <div className="space-y-2 p-4">
   484→            {[0, 1, 2].map((i) => (
   485→              <Skeleton key={i} className="h-10" />
   486→            ))}
   487→          </div>
   488→        ) : (
   489→          <Table>
   490→            <TableHeader>
   491→              <TableRow>
   492→                <TableHead>Name</TableHead>
   493→                <TableHead>Category</TableHead>
   494→                <TableHead>Price</TableHead>
   495→                <TableHead>Daily</TableHead>
   496→                <TableHead>Status</TableHead>
   497→                <TableHead className="text-right">Actions</TableHead>
   498→              </TableRow>
   499→            </TableHeader>
   500→            <TableBody>
   501→              {products.length === 0 ? (
   502→                <TableRow>
   503→                  <TableCell colSpan={6} className="py-8 text-center text-muted-foreground">
   504→                    No products yet.
   505→                  </TableCell>
   506→                </TableRow>
   507→              ) : (
   508→                products.map((p) => (
   509→                  <TableRow key={p.id}>
   510→                    <TableCell className="font-medium">{p.name}</TableCell>
   511→                    <TableCell>{p.category}</TableCell>
   512→                    <TableCell>{formatETB(p.price)}</TableCell>
   513→                    <TableCell>{formatETB(p.dailyIncome)}/day</TableCell>
   514→                    <TableCell>
   515→                      <StatusBadge
   516→                        label={p.status}
   517→                        tone={p.status === "AVAILABLE" ? "green" : "muted"}
   518→                      />
   519→                    </TableCell>
   520→                    <TableCell className="text-right">
   521→                      <div className="flex justify-end gap-1">
   522→                        <Button
   523→                          size="icon"
   524→                          variant="ghost"
   525→                          aria-label="Edit"
   526→                          onClick={() => openEdit(p)}
   527→                        >
   528→                          <Pencil className="size-4" />
   529→                        </Button>
   530→                        <Button
   531→                          size="icon"
   532→                          variant="ghost"
   533→                          aria-label="Delete"
   534→                          disabled={deletingId === p.id}
   535→                          onClick={() => remove(p.id)}
   536→                        >
   537→                          {deletingId === p.id ? (
   538→                            <Loader2 className="size-4 animate-spin" />
   539→                          ) : (
   540→                            <Trash2 className="size-4 text-rose-600" />
   541→                          )}
   542→                        </Button>
   543→                      </div>
   544→                    </TableCell>
   545→                  </TableRow>
   546→                ))
   547→              )}
   548→            </TableBody>
   549→          </Table>
   550→        )}
   551→      </Card>
   552→
   553→      <ProductDialog
   554→        open={dialogOpen}
   555→        onOpenChange={setDialogOpen}
   556→        initial={editing}
   557→        onSaved={load}
   558→      />
   559→    </div>
   560→  );
   561→}

   564→function PurchasesTab() {
   565→  const [items, setItems] = useState<PurchasePublic[]>([]);
   566→  const [loading, setLoading] = useState(true);
   567→  const [busyId, setBusyId] = useState<string | null>(null);

   570→    setLoading(true);
   571→    try {
   572→      const res = await purchasesApi.list(true);
   573→      setItems(res.purchases);
   574→    } catch (err) {
   575→      toast.error(err instanceof Error ? err.message : "Failed to load purchases");
   576→    } finally {
   577→      setLoading(false);
   578→    }
   579→  };
   580→
   581→  useEffect(() => {
   582→    void load();
   583→  }, []);
   584→
   585→  const act = async (id: string, kind: "approve" | "reject") => {
   586→    setBusyId(id);
   587→    try {
   588→      if (kind === "approve") await purchasesApi.approve(id);
   589→      else await purchasesApi.reject(id);
   590→      toast.success(kind === "approve" ? "Purchase approved" : "Purchase rejected");
   591→      await load();
   592→    } catch (err) {
   593→      toast.error(err instanceof Error ? err.message : "Action failed");
   594→    } finally {
   595→      setBusyId(null);
   596→    }
   597→  };
   598→
   599→  return (
   600→    <Card className="p-0">
   601→      {loading ? (
   602→        <div className="space-y-2 p-4">
   603→          {[0, 1, 2].map((i) => (
   604→            <Skeleton key={i} className="h-10" />
   605→          ))}
   606→        </div>
   607→      ) : (
   608→        <Table>
